#include <Wire.h>

#define SDA_PIN D2 // GPIO4
#define SCL_PIN D1 // GPIO5
const int MPU_ADDR = 0x68;

// Nyers adatok
int16_t raw_acc_x, raw_acc_y, raw_acc_z;
int16_t raw_gyro_x, raw_gyro_y, raw_gyro_z;

// Giroszkóp nullpont eltolódás (offset)
float gyro_x_offset = 0, gyro_y_offset = 0, gyro_z_offset = 0;

// Kiszámított szögek
float pitch = 0, roll = 0, yaw = 0;

// Időzítés a giroszkóp integrálásához
unsigned long last_time = 0;

void setup() {
  Serial.begin(115200);
  delay(1000);

  Wire.begin(SDA_PIN, SCL_PIN);
  Wire.setClock(100000);

  // MPU ébresztése
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x6B); // PWR_MGMT_1 regiszter
  Wire.write(0);    
  if (Wire.endTransmission() != 0) {
    Serial.println("Hiba: MPU nem válaszol! Ellenőrizd a bekötést.");
    while (1);
  }

  Serial.println("MPU-9250/6500 kalibrálása... Ne mozgasd a szenzort!");
  calibrateGyro();
  Serial.println("Kalibráció kész! Adatok olvasása...");
  
  last_time = millis();
}

void loop() {
  // 1. Nyers adatok beolvasása
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x3B); 
  Wire.endTransmission(false);
  Wire.requestFrom(MPU_ADDR, 14, true);

  if (Wire.available() == 14) {
    raw_acc_x  = Wire.read() << 8 | Wire.read();
    raw_acc_y  = Wire.read() << 8 | Wire.read();
    raw_acc_z  = Wire.read() << 8 | Wire.read();
    Wire.read(); Wire.read(); // Hőmérséklet átugrása
    raw_gyro_x = Wire.read() << 8 | Wire.read();
    raw_gyro_y = Wire.read() << 8 | Wire.read();
    raw_gyro_z = Wire.read() << 8 | Wire.read();

    // 2. Időkülönbség számítása (másodpercben)
    unsigned long current_time = millis();
    float dt = (current_time - last_time) / 1000.0;
    last_time = current_time;

    // 3. Giroszkóp szögsebesség átszámítása (°/s-ba) és offset korrekció
    // FS_SEL=0 esetén a skálázási tényező: 131.0 LSB/(°/s)
    float gyro_x_rate = (raw_gyro_x - gyro_x_offset) / 131.0;
    float gyro_y_rate = (raw_gyro_y - gyro_y_offset) / 131.0;
    float gyro_z_rate = (raw_gyro_z - gyro_z_offset) / 131.0;

    // 4. Gyorsulásmérőből számított szögek (fokokban)
    // AFS_SEL=0 esetén a skálázási tényező: 16384 LSB/g
    float ax = raw_acc_x / 16384.0;
    float ay = raw_acc_y / 16384.0;
    float az = raw_acc_z / 16384.0;

    float acc_pitch = atan2(-ax, sqrt(ay * ay + az * az)) * 180.0 / M_PI;
    float acc_roll  = atan2(ay, az) * 180.0 / M_PI;

    // 5. Komplementer szűrő (96% giroszkóp integrálás + 4% gyorsulásmérő korrekció)
    pitch = 0.96 * (pitch + gyro_x_rate * dt) + 0.04 * acc_pitch;
    roll  = 0.96 * (roll  + gyro_y_rate * dt) + 0.04 * acc_roll;
    
    // Yaw kizárólag giroszkópból integrálható (magnetométer hiányában lassan driftelhet)
    yaw += gyro_z_rate * dt;

    // 6. Kiírás a Soros Monitorra
    Serial.print("Pitch (Dőlés): "); Serial.print(pitch, 2);
    Serial.print("° | Roll (Gördülés): "); Serial.print(roll, 2);
    Serial.print("° | Yaw (Elfordulás): "); Serial.print(yaw, 2);
    Serial.println("°");
  }

  delay(20); // Kb. 50 Hz-es frissítési frekvencia
}

// Giroszkóp nullpont kalibrációs rutin
void calibrateGyro() {
  long sum_x = 0, sum_y = 0, sum_z = 0;
  int samples = 500;

  for (int i = 0; i < samples; i++) {
    Wire.beginTransmission(MPU_ADDR);
    Wire.write(0x43); // Gyro adatok kezdete
    Wire.endTransmission(false);
    Wire.requestFrom(MPU_ADDR, 6, true);

    if (Wire.available() == 6) {
      sum_x += Wire.read() << 8 | Wire.read();
      sum_y += Wire.read() << 8 | Wire.read();
      sum_z += Wire.read() << 8 | Wire.read();
    }
    delay(3);
  }

  gyro_x_offset = sum_x / (float)samples;
  gyro_y_offset = sum_y / (float)samples;
  gyro_z_offset = sum_z / (float)samples;
}
